library ieee;
use ieee.std_logic_1164.all;

entity forLoop is 
	port (
		  a : in  std_logic_vector(7 downto 0);
		  p : inout  std_logic_vector(7 downto 0);
        y : out std_logic
	);
end forLoop;

architecture code of forLoop is
	begin
			process(a)
				begin
					p(0) <= a(0);
						for i in 1 to 7 loop
							p(i) <= p(i-1) xor a(i);
						end loop;
					y <= p(7);
			end process;
end code;

configuration c_forLoop of forLoop is
    for code end for;
end configuration;
